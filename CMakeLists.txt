cmake_minimum_required(VERSION 3.24)
project(Scratcher VERSION 0.8.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _SILENCE_CXX20_U8PATH_DEPRECATION_WARNING __x86_64__)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/zlib EXCLUDE_FROM_ALL)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZYDIS_FEATURE_ENCODER OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/zydis EXCLUDE_FROM_ALL)
set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CMAKE_SYSTEM_PROCESSOR AMD64)
set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_CSTEST OFF CACHE BOOL "" FORCE)
set(CAPSTONE_INSTALL OFF CACHE BOOL "" FORCE)
set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
set(CAPSTONE_ARM_SUPPORT ON CACHE BOOL "" FORCE)
set(CAPSTONE_ARM64_SUPPORT ON CACHE BOOL "" FORCE)
set(CAPSTONE_MIPS_SUPPORT ON CACHE BOOL "" FORCE)
set(CAPSTONE_RISCV_SUPPORT ON CACHE BOOL "" FORCE)
set(BUILD_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/capstone EXCLUDE_FROM_ALL)
set(GH vendor/ghidra/Ghidra/Features/Decompiler/src/decompile/cpp)
set(GH_NAMES xml marshal space float address pcoderaw translate opcodes globalcontext
 capability architecture options graph cover block cast typeop database cpool comment
 stringmanage modelrules fspec action loadimage grammar varnode op type variable varmap
 jumptable emulate emulateutil flow userop expression multiprecision funcdata funcdata_block
 funcdata_op funcdata_varnode unionresolve pcodeinject heritage prefersplit rangeutil ruleaction
 subflow blockaction merge double transform constseq bitfield coreaction condexe override
 dynamic crc32 prettyprint printlanguage printc printjava memstate opbehavior paramid signature
 sleigh pcodeparse pcodecompile sleighbase slghsymbol slghpatexpress slghpattern semantics
 context slaformat compression filemanage sleigh_arch inject_sleigh libdecomp)
foreach(name IN LISTS GH_NAMES)
 list(APPEND GH_SOURCES ${GH}/${name}.cc)
endforeach()
add_library(ghidra_native STATIC ${GH_SOURCES})
target_include_directories(ghidra_native SYSTEM PUBLIC ${GH})
target_include_directories(ghidra_native PRIVATE vendor/zlib ${CMAKE_BINARY_DIR}/vendor/zlib)
target_link_libraries(ghidra_native PUBLIC zlibstatic)
target_link_options(ghidra_native INTERFACE /WHOLEARCHIVE:$<TARGET_FILE:ghidra_native>)
target_compile_options(ghidra_native PRIVATE /O2 /w /bigobj /MP6)
set(SCRATCHER_CUDA_ROOT "${CMAKE_SOURCE_DIR}/vendor/cuda-runtime" CACHE PATH "Pinned optional CUDA headers and runtime compiler")
set(SCRATCHER_CUDA_DEFAULT OFF)
if(EXISTS "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_runtime/include/cuda.h")
 set(SCRATCHER_CUDA_DEFAULT ON)
endif()
option(SCRATCHER_ENABLE_CUDA "Build optional dynamically loaded CUDA similarity acceleration" ${SCRATCHER_CUDA_DEFAULT})
add_library(hyperdecomp_accel STATIC src/accel/matcher.cpp)
target_include_directories(hyperdecomp_accel PUBLIC src)
target_compile_options(hyperdecomp_accel PRIVATE /O2 /W4 /EHsc /utf-8)
if(SCRATCHER_ENABLE_CUDA)
 if(NOT EXISTS "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_runtime/include/cuda.h" OR NOT EXISTS "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/include/nvrtc.h")
  message(FATAL_ERROR "CUDA headers missing. Run scripts/setup-hyperdecomp.ps1 or configure -DSCRATCHER_ENABLE_CUDA=OFF.")
 endif()
 target_compile_definitions(hyperdecomp_accel PRIVATE SCRATCHER_ENABLE_CUDA=1)
 target_include_directories(hyperdecomp_accel SYSTEM PRIVATE
  "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_runtime/include"
  "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/include")
 add_custom_target(hyperdecomp_cuda ALL
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/$<CONFIG>/cuda"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
   "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/bin/nvrtc64_120_0.dll"
   "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/bin/nvrtc-builtins64_124.dll"
   "${CMAKE_BINARY_DIR}/$<CONFIG>/cuda")
 add_dependencies(hyperdecomp_accel hyperdecomp_cuda)
 install(FILES
  "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/bin/nvrtc64_120_0.dll"
  "${SCRATCHER_CUDA_ROOT}/nvidia/cuda_nvrtc/bin/nvrtc-builtins64_124.dll" DESTINATION cuda)
 install(FILES "${SCRATCHER_CUDA_ROOT}/nvidia_cuda_nvrtc_cu12-12.4.127.dist-info/License.txt" DESTINATION licenses/cuda-nvrtc)
 install(FILES "${SCRATCHER_CUDA_ROOT}/nvidia_cuda_runtime_cu12-12.4.127.dist-info/License.txt" DESTINATION licenses/cuda-headers)
endif()
file(GLOB FEATURE_SOURCES CONFIGURE_DEPENDS src/feature_*.cpp)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS vendor/lua/src/*.c)

set_source_files_properties(${LUA_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(scratcher_lua STATIC ${LUA_SOURCES})

target_compile_options(scratcher_lua PRIVATE /EHsc /w /MP6)
add_library(scratcher_core STATIC src/binary.cpp src/engine.cpp src/mcp.cpp src/features.cpp src/tools_extra.cpp ${FEATURE_SOURCES})
target_include_directories(scratcher_core PUBLIC src)
target_include_directories(scratcher_core SYSTEM PUBLIC vendor/json)
target_include_directories(scratcher_core PRIVATE vendor/zlib ${CMAKE_BINARY_DIR}/vendor/zlib)
target_link_libraries(scratcher_core PUBLIC ghidra_native Zydis capstone_static scratcher_lua hyperdecomp_accel bcrypt advapi32 dbghelp)
target_include_directories(scratcher_core SYSTEM PUBLIC vendor/capstone/include)
target_compile_options(scratcher_core PRIVATE /W4 /utf-8 /EHsc /MP6)

 vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
 vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)

target_compile_options(HyperDecompBench PRIVATE /O2 /EHsc /utf-8)
add_executable(scratcher_tests tests/core_tests.cpp)
target_link_libraries(scratcher_tests PRIVATE scratcher_core)

add_executable(ScratcherDebugFixture examples/debug_fixture.cpp)
add_executable(ScratcherAnalysisFixture examples/analysis_fixture.cpp)

target_link_options(ScratcherAnalysisFixture PRIVATE /guard:cf /DELAYLOAD:version.dll)

 add_executable(ScratcherDeepFixture${deep_variant} examples/deep_analysis_fixture.cpp)
 target_compile_options(ScratcherDeepFixture${deep_variant} PRIVATE /${deep_variant} /Zi /EHsc)
 target_link_options(ScratcherDeepFixture${deep_variant} PRIVATE /DEBUG /OPT:NOICF)
 install(TARGETS ScratcherDeepFixture${deep_variant} RUNTIME DESTINATION samples)
endforeach()

 ${CMAKE_SOURCE_DIR}/data ${CMAKE_BINARY_DIR}/$<CONFIG>/data)
add_dependencies(Scratcher scratcher_data)
add_dependencies(ScratcherMCP scratcher_data)
add_dependencies(HyperDecompCLI scratcher_data)
add_dependencies(scratcher_tests scratcher_data)
add_test(NAME native_core COMMAND scratcher_tests)
file(GLOB FEATURE_TEST_SOURCES CONFIGURE_DEPENDS tests/*_tests.cpp)
foreach(test_source IN LISTS FEATURE_TEST_SOURCES)
 get_filename_component(test_name ${test_source} NAME_WE)
 if(NOT test_name STREQUAL "core_tests")
  add_executable(${test_name} ${test_source})
  if(test_name STREQUAL "accelerator_tests")
   target_link_libraries(${test_name} PRIVATE hyperdecomp_accel)
  else()
   target_link_libraries(${test_name} PRIVATE scratcher_core)
  endif()
  add_test(NAME ${test_name} COMMAND ${test_name})
  add_dependencies(${test_name} scratcher_data)
  if(test_name STREQUAL "advanced_symbols_tests")
   target_compile_options(${test_name} PRIVATE /Zi)
   target_link_options(${test_name} PRIVATE /DEBUG)
  endif()
  if(test_name STREQUAL "advanced_debugger_tests")
   add_dependencies(${test_name} ScratcherDebugFixture)
  endif()
  if(test_name STREQUAL "advanced_sdk_tests")
   add_dependencies(${test_name} ScratcherPluginSample)
  endif()
  if(test_name STREQUAL "ui_workspace_tests")
   target_sources(${test_name} PRIVATE src/ui_workspace.cpp src/ui_extra.cpp)
   target_link_libraries(${test_name} PRIVATE imgui comdlg32)
  endif()
  if(test_name STREQUAL "decompiler_switch_tests")
   add_dependencies(${test_name} ScratcherAnalysisFixture)
  endif()
 endif()
endforeach()
